create_artifact_black_hands_dagger_effect = {
	# Get the character the artifact is being made for.
	$OWNER$ = { save_scope_as = owner }
	# Not really used, but if we don't set the scopes we get errors in the feature selection
	set_artifact_rarity_famed = yes

	# Create the artifact
	create_artifact = {
		name = artifact_black_hands_dagger_name
		description = artifact_black_hands_dagger_description
		type = dagger
		visuals = black_hands_dagger
		wealth = scope:wealth
		quality = scope:quality
		history = {
				type = discovered
				actor = scope:inspiration_owner
				recipient = root
				location = scope:inspiration_owner.location
		}
		modifier = black_hands_dagger_modifier
		save_scope_as = newly_created_artifact
	}

	scope:newly_created_artifact = {
		save_scope_as = epic
		save_scope_value_as = {
			name = dagger
			value = yes
		}
		
	}
}
